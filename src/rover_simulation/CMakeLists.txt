cmake_minimum_required(VERSION 3.16)
project(rover_simulation)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz_transport_vendor REQUIRED)
find_package(gz-transport13 REQUIRED)
add_library(rover_torque_drive SHARED src/torque_drive.cpp)
target_compile_features(rover_torque_drive PUBLIC cxx_std_17)
target_compile_options(rover_torque_drive PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(rover_torque_drive gz-sim8::gz-sim8 gz-plugin2::register gz-transport13::gz-transport13)
install(TARGETS rover_torque_drive LIBRARY DESTINATION lib)
add_library(rover_differential SHARED src/differential.cpp)
target_compile_features(rover_differential PUBLIC cxx_std_17)
target_compile_options(rover_differential PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(rover_differential gz-sim8::gz-sim8 gz-plugin2::register)
install(TARGETS rover_differential LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_differential test/test_differential.cpp)
  target_compile_features(test_differential PRIVATE cxx_std_17)
  add_test(NAME differential_law COMMAND test_differential)
  add_executable(test_motor test/test_motor.cpp)
  target_compile_features(test_motor PUBLIC cxx_std_17)
  add_test(NAME motor_law COMMAND test_motor)
endif()
ament_package()
